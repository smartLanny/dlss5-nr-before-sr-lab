cmake_minimum_required(VERSION 3.21)
project(XiaofengQuickbar LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use an x64 MSVC Windows build. Do not mix MinGW ImGui struct-return ABIs.")
endif()
set(RESHADE_SOURCE "" CACHE PATH "Pinned ReShade v6.8.0 checkout (including deps/imgui)")
if(NOT EXISTS "${RESHADE_SOURCE}/include/reshade.hpp" OR NOT EXISTS "${RESHADE_SOURCE}/deps/imgui/imgui.h")
  message(FATAL_ERROR "Missing ReShade or its ImGui submodule")
endif()
add_library(quickbar SHARED quickbar/quickbar.cpp)
target_compile_features(quickbar PRIVATE cxx_std_17)
target_compile_definitions(quickbar PRIVATE UNICODE _UNICODE)
target_compile_options(quickbar PRIVATE /utf-8 /W4 /permissive- /EHsc)
target_include_directories(quickbar PRIVATE "${RESHADE_SOURCE}/include" "${RESHADE_SOURCE}/deps/imgui")
target_link_libraries(quickbar PRIVATE user32 kernel32 psapi)
set_target_properties(quickbar PROPERTIES PREFIX "" OUTPUT_NAME "zjz-quickbar" SUFFIX ".addon64" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(frame_history_test tests/frame_history_test.cpp)
target_compile_features(frame_history_test PRIVATE cxx_std_17)
target_compile_options(frame_history_test PRIVATE /UNDEBUG)
enable_testing()
add_test(NAME frame_history COMMAND frame_history_test)
