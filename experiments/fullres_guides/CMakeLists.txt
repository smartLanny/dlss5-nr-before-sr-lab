cmake_minimum_required(VERSION 3.20)
project(nr_fullres_guides LANGUAGES CXX)
include(CTest)
add_executable(fullres_policy_tests test_policy.cpp)
target_compile_features(fullres_policy_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(fullres_policy_tests PRIVATE /W4 /WX)
else()
  target_compile_options(fullres_policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME fullres_policy COMMAND fullres_policy_tests)
option(NR_OF_BUILD_ADDON "Build unverified Windows D3D12 input-probe addon" OFF)
if(NR_OF_BUILD_ADDON)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Addon prototype requires MSVC x64 Windows; portable policy is separate")
  endif()
  set(FFX_ROOT "" CACHE PATH "Official FidelityFX SDK v1.1.4 checkout")
  set(FFX_LIBRARIES "" CACHE STRING "Explicit semicolon-separated full paths to prebuilt SDK opticalflow/DX12 dependency libraries")
  set(NVOF_INCLUDE "" CACHE PATH "Official Optical Flow SDK NvOFInterface directory")
  set(RESHADE_INCLUDE "" CACHE PATH "Headers matching the ReShade runtime under test")
  set(IMGUI_INCLUDE "" CACHE PATH "ImGui headers matching that ReShade runtime")
  foreach(required IN ITEMS "${FFX_ROOT}/sdk/include/FidelityFX/host/ffx_opticalflow.h"
      "${NVOF_INCLUDE}/nvOpticalFlowD3D12.h" "${RESHADE_INCLUDE}/reshade.hpp" "${IMGUI_INCLUDE}/imgui.h")
    if(NOT EXISTS "${required}")
      message(FATAL_ERROR "Missing explicit SDK input: ${required}")
    endif()
  endforeach()
  if(NOT FFX_LIBRARIES)
    message(FATAL_ERROR "FFX_LIBRARIES must name compiled official SDK libraries, including their dependencies; no stub backend is substituted")
  endif()
  foreach(lib IN LISTS FFX_LIBRARIES)
    if(NOT EXISTS "${lib}")
      message(FATAL_ERROR "Missing compiled SDK library: ${lib}")
    endif()
  endforeach()
  add_library(nr_fullres_guides MODULE addon.cpp backend.cpp)
  target_compile_features(nr_fullres_guides PRIVATE cxx_std_17)
  target_compile_definitions(nr_fullres_guides PRIVATE RESHADE_ADDON=1)
  target_compile_options(nr_fullres_guides PRIVATE /W4 /utf-8 /EHsc)
  target_include_directories(nr_fullres_guides PRIVATE "${FFX_ROOT}/sdk/include" "${NVOF_INCLUDE}" "${RESHADE_INCLUDE}" "${IMGUI_INCLUDE}")
  target_link_libraries(nr_fullres_guides PRIVATE ${FFX_LIBRARIES} d3d12 dxgi d3dcompiler)
  set_target_properties(nr_fullres_guides PROPERTIES PREFIX "" SUFFIX ".addon64")
  # No Core package, NVIDIA runtime, Manager payload, installer or automatic deployment.
endif()
