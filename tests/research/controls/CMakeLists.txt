cmake_minimum_required(VERSION 3.16)
project(nr_controls_research LANGUAGES CXX)
enable_testing()
add_executable(nr_controls_test test.cpp)
target_compile_features(nr_controls_test PRIVATE cxx_std_17)
target_include_directories(nr_controls_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
if(MSVC)
  target_compile_options(nr_controls_test PRIVATE /W4 /WX)
else()
  target_compile_options(nr_controls_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME nr_controls_contract COMMAND nr_controls_test)
